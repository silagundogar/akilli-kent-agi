Çoklu cihaz, gerçek zamanlı senkronizasyon ve bulut veritabanı

Şu An Yerel Hafıza (LocalStorage) Kullanılıyor

Farklı telefon ve bilgisayarların aynı verileri anlık olarak görebilmesi için aşağıdaki alana Supabase bilgilerinizi giriniz.

Supabase Dashboard > Settings > API > Project URL
Supabase Dashboard > Settings > API > Project API Keys > `anon public`
1 Dakikada Supabase Kurulumu & SQL Tablosu
  1. supabase.com adresinde ücretsiz proje açın.
  2. Sol menüden SQL Editor bölümüne gidin.
  3. Aşağıdaki SQL kodunu yapıştırıp yeşil RUN butonuna basın.
  4. Project Settings > API kısmındaki URL ve Anon Key'i yukarıya yapıştırıp "Kaydet"e basın!
create table if not exists public.issues (
  id text primary key,
  code text unique not null,
  title text not null,
  "desc" text,
  category text,
  department text,
  urgency text default 'Normal',
  status text default 'Yeni',
  neighborhood text,
  address text,
  lat double precision,
  lng double precision,
  "citizenName" text,
  "citizenPhone" text,
  "isAnonymous" boolean default false,
  "createdAt" text,
  "assignedTeam" text,
  "fieldNote" text,
  "imgBefore" text,
  "imgAfter" text,
  rating integer,
  feedback text,
  history jsonb default '[]'::jsonb
);
alter table public.issues enable row level security;
create policy "Herkes ihbarları görebilir" on public.issues for select using (true);
create policy "Herkes yeni ihbar ekleyebilir" on public.issues for insert with check (true);
create policy "Herkes ihbarları güncelleyebilir" on public.issues for update using (true);
create policy "Herkes ihbar silebilir" on public.issues for delete using (true);
alter publication supabase_realtime add table public.issues;
